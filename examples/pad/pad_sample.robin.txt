    SET SmArgs TO $'''%SmExe% --browser %Browser% --browser-version stable --output json'''
    IF UseProxy THEN
        SET SmArgs TO $'''%SmArgs% --proxy %ProxyAddr%'''
    END
    Scripting.RunDOSCommand.RunDOSCommandAndFailOnTimeout DOSCommandOrApplication: SmArgs WorkingDirectory: BaseDir Timeout: 300 StandardOutput=> SmOutput StandardError=> SmError ExitCode=> SmExit
    # 終了コードを先に見る。exe が起動できないと出力が空になり、そのまま
    # JSON に変換しようとして「CustomObject が取れません」という、原因の
    # 分からないエラーになる。
    IF SmExit <> 0 THEN
        SET Halt TO True
        SET HaltReason TO $'''Selenium Manager を実行できませんでした'''
        Display.ShowMessageDialog.ShowMessage Title: $'''Selenium Manager を実行できません''' Message: SmError Icon: Display.Icon.Warning Buttons: Display.Buttons.OK DefaultButton: Display.DefaultButton.Button1 IsTopMost: True ButtonPressed=> SmBtn2
    END
    IF Halt = False THEN
    Variables.ConvertJsonToCustomObject Json: SmOutput CustomObject=> SmObj
    IF SmObj['result']['code'] = 0 THEN
        SET DriverExe TO SmObj['result']['driver_path']
        SET DriverSrc TO $'''Selenium Manager（ブラウザーのバージョンに合わせて自動取得）'''
        SET SmMsg TO SmObj['result']['message']
    END
    IF SmObj['result']['code'] <> 0 THEN
        SET Halt TO True
        SET HaltReason TO $'''ドライバーの取得に失敗しました（Selenium Manager）'''
        Display.ShowMessageDialog.ShowMessage Title: $'''ドライバー取得に失敗''' Message: SmOutput Icon: Display.Icon.Information Buttons: Display.Buttons.OK DefaultButton: Display.DefaultButton.Button1 IsTopMost: True ButtonPressed=> SmBtn
    END
    END
END

# --- 起動したブラウザーとドライバーを確認する ---

# 起動したのが要求どおりのブラウザーかを先に確かめる。ドライバーは取り違えても

# ---------- 起点画面かどうかを確認する ----------
# ループ最初の操作対象がこの画面にあるかを、クリックせずに調べる。
# 見つからなければ、その場で進めてもらって何度でも試せるようにする。
# 1 度きりで中止すると、画面を直してから実行し直すことになって手間が増える。
SET StartOk TO False
SET StartTry TO 0
LOOP WHILE StartOk = False
    # ---------- 起点画面まで進む ----------
    # ログイン直後の画面から、繰り返しの起点までメニューを辿る。
    # 毎周やり直すのは、1 回目にログイン前の画面で空振りしても、
    # ログインしてから[OK]を押せばそのまま起点まで進めるようにするため。
    # すでに起点にいるときはメニューが無くて空振りするだけ。
    # click #btn-signin
    SET ActBody TO $'''{"script": "%JsAct%", "args": [["#btn-signin", "aria/Sign in"], "click", ""]}'''
    Web.InvokeWebService.InvokeWebService Url: ExecUrl Method: Web.Method.Post Accept: AppJson ContentType: AppJson RequestBody: ActBody EncodeRequestBody: False FailOnErrorStatus: False Response=> ActResp StatusCode=> ActStatus
    WAIT 1
    # click #menu-orders
    SET ActBody TO $'''{"script": "%JsAct%", "args": [["#menu-orders", "aria/Orders"], "click", ""]}'''
    Web.InvokeWebService.InvokeWebService Url: ExecUrl Method: Web.Method.Post Accept: AppJson ContentType: AppJson RequestBody: ActBody EncodeRequestBody: False FailOnErrorStatus: False Response=> ActResp StatusCode=> ActStatus
    WAIT 1

    # 起点かどうかを調べる。移動で ActBody を使い回しているので、
    # ここで確認用の本文に入れ直す。
    SET ActBody TO $'''{"script": "%JsAct%", "args": [["#search-key", "aria/Order key"], "find", ""]}'''
    Web.InvokeWebService.InvokeWebService Url: ExecUrl Method: Web.Method.Post Accept: AppJson ContentType: AppJson RequestBody: ActBody EncodeRequestBody: False FailOnErrorStatus: False Response=> ActResp StatusCode=> ActStatus
    # 応答コードを先に見る。エラー応答には ok が無いので、そのまま
    # 読みに行くと「プロパティがありません」で止まる。
    IF ActStatus = 200 THEN
        Variables.ConvertJsonToCustomObject Json: ActResp CustomObject=> ActObj
        IF ActObj['value']['ok'] = True THEN
            SET StartOk TO True
        END
    END
    IF StartOk = False THEN
        SET StartTry TO StartTry + 1
        IF StartTry = 1 THEN
            Display.ShowMessageDialog.ShowMessage Title: $'''ログインは終わりましたか''' Message: $'''ログインが済んでいなければ、ログインしてから[OK]。済んでいれば「Order key」が見える画面まで進めて[OK]。''' Icon: Display.Icon.Warning Buttons: Display.Buttons.OKCancel DefaultButton: Display.DefaultButton.Button1 IsTopMost: True ButtonPressed=> StartBtn
        END
        IF StartTry > 1 THEN
            Display.ShowMessageDialog.ShowMessage Title: $'''起点画面ではありません''' Message: $'''起点画面が出ていません。「Order key」が見える画面まで進めて[OK]。（%StartTry% 回目）''' Icon: Display.Icon.Warning Buttons: Display.Buttons.OKCancel DefaultButton: Display.DefaultButton.Button1 IsTopMost: True ButtonPressed=> StartBtn
        END
        IF StartBtn = $'''Cancel''' THEN
            SET Halt TO True
            SET HaltReason TO $'''繰り返しの起点画面に到達できませんでした'''
            SET StartOk TO True
        END
    END
END

            # これを外すと上限を刻んで回したとき続きが流せない。
            SET DoRow TO False
            IF Row['結果'] = $'''失敗''' THEN
                SET DoRow TO True
            END
            IF Row['結果'] = $'''未実行''' THEN
                SET DoRow TO True
            END
            IF DoRow = False THEN
                NEXT LOOP
            END
        END

        # skip 列に値がある行は飛ばす（再実行 CSV には skip 列が無いので除外）
        IF RetryMode = False THEN

                SET SkipCount TO SkipCount + 1
                File.WriteText File: ResultFile TextToWrite: $'''%RowId%,%RowKey%,スキップ,,,''' AppendNewLine: True IfFileExists: File.IfFileExists.Append
                NEXT LOOP
            END
        END

        IF Attempted >= MaxItems THEN
            File.WriteText File: ResultFile TextToWrite: $'''%RowId%,%RowKey%,未実行,"件数上限 %MaxItems% 件に達したため",,''' AppendNewLine: True IfFileExists: File.IfFileExists.Append
            NEXT LOOP

            WAIT 1
