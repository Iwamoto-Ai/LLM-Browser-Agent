# ---------- セットアップ失敗を検知して止める ----------
# ここで止めないと、ループ先頭で RowError がリセットされるため、

# ================= 明細（CSV）を読み込む =================
IF Halt = False THEN
    File.ReadFromCSVFile.ReadCSV CSVFile: DetailsFile Encoding: File.CSVEncoding.UTF8 TrimFields: True FirstLineContainsColumnNames: True ColumnsSeparator: File.CSVColumnsSeparator.Comma CSVTable=> Rows
    # 結果 CSV は、そのまま明細として読み直せる列構成にする。
    # ID だけでなく業務キーも入れないと、再実行で対象を特定できない。
    File.WriteText File: ResultFile TextToWrite: $'''ID,KEY,結果,理由,エビデンス,実行日時''' AppendNewLine: True IfFileExists: File.IfFileExists.Overwrite
    SET OkCount TO 0
    SET NgCount TO 0
    SET SkipCount TO 0
    SET Attempted TO 0
    SET PrevFailed TO False

    # ================= 明細ごとの繰り返し =================
    LOOP FOREACH Row IN Rows
        # 明細の列を変数に取り出す（リテラル内に単引用符を入れないため）
        SET Col1 TO Row['ID']
        SET Col2 TO Row['KEY']
        SET RowId TO Col1
        SET RowKey TO Col2
        SET RowError TO $''''''

        # 再実行モードでは「失敗」行だけを対象にする
        IF RetryMode THEN
            # 対象は「失敗」と「未実行」。未実行は件数上限で打ち切った行で、
            # これを外すと上限を刻んで回したとき続きが流せない。
            SET DoRow TO False
            IF Row['結果'] = $'''失敗''' THEN
                SET DoRow TO True
            END
            IF Row['結果'] = $'''未実行''' THEN
                SET DoRow TO True
            END
            IF DoRow = False THEN
                NEXT LOOP
            END
        END

        # skip 列に値がある行は飛ばす（再実行 CSV には skip 列が無いので除外）
        IF RetryMode = False THEN
            IF Row['skip'] <> $'''''' THEN
                SET SkipCount TO SkipCount + 1
                File.WriteText File: ResultFile TextToWrite: $'''%RowId%,%RowKey%,スキップ,,,''' AppendNewLine: True IfFileExists: File.IfFileExists.Append
                NEXT LOOP
            END
        END

        # 件数上限。打ち切った行も記録に残す（どこから再開するか分かる）
        IF Attempted >= MaxItems THEN
            File.WriteText File: ResultFile TextToWrite: $'''%RowId%,%RowKey%,未実行,"件数上限 %MaxItems% 件に達したため",,''' AppendNewLine: True IfFileExists: File.IfFileExists.Append
            NEXT LOOP
        END
        SET Attempted TO Attempted + 1

        # 直前の件が失敗していたら、起点に復帰してから始める。
        # 失敗した画面のまま次の件を始めると 1 件の失敗が全件に連鎖する。
        IF PrevFailed THEN
            # 復帰: click #home-link
            SET ActBody TO $'''{"script": "%JsAct%", "args": [["#home-link", "aria/Home"], "click", ""]}'''
            Web.InvokeWebService.InvokeWebService Url: ExecUrl Method: Web.Method.Post Accept: AppJson ContentType: AppJson RequestBody: ActBody EncodeRequestBody: False FailOnErrorStatus: False Response=> ActResp StatusCode=> ActStatus
            WAIT 1
            # 復帰: click #menu-orders
            SET ActBody TO $'''{"script": "%JsAct%", "args": [["#menu-orders", "aria/Orders"], "click", ""]}'''
            Web.InvokeWebService.InvokeWebService Url: ExecUrl Method: Web.Method.Post Accept: AppJson ContentType: AppJson RequestBody: ActBody EncodeRequestBody: False FailOnErrorStatus: False Response=> ActResp StatusCode=> ActStatus
            WAIT 1
            SET PrevFailed TO False
        END
        # 悲観的に置く。最後まで通れば False に戻る（フラグを立てる箇所が 1 行で済む）
        SET PrevFailed TO True

        # 日時つきのファイル名を作る。MM は月、mm は分。
        DateTime.GetCurrentDateTime.Local DateTimeFormat: DateTime.DateTimeFormat.DateAndTime CurrentDateTime=> NowDt
        Text.ConvertDateTimeToText.FromCustomDateTime DateTime: NowDt CustomFormat: $'''yyyyMMdd_HHmmss''' Result=> Stamp
        Text.ConvertDateTimeToText.FromCustomDateTime DateTime: NowDt CustomFormat: $'''yyyy/MM/dd HH:mm:ss''' Result=> RecStamp
        SET ShotPath TO $'''%ShotDir%\\%RowId%__%RowKey%__%Stamp%.png'''
        SET FailShot TO $'''%ShotDir%\\fail__%RowId%__%RowKey%__%Stamp%.png'''
        File.WriteText File: LogFile TextToWrite: $'''[%RecStamp%] %RowId% / %RowKey% 開始''' AppendNewLine: True IfFileExists: File.IfFileExists.Append

        # 💬 注文 %Col2%（%Col1%）の確認を開始
        # [1] fill #search-key
        SET ActBody TO $'''{"script": "%JsAct%", "args": [["#search-key", "aria/Order key"], "fill", "%Col2%"]}'''
        Web.InvokeWebService.InvokeWebService Url: ExecUrl Method: Web.Method.Post Accept: AppJson ContentType: AppJson RequestBody: ActBody EncodeRequestBody: False FailOnErrorStatus: False Response=> ActResp StatusCode=> ActStatus
        Variables.ConvertJsonToCustomObject Json: ActResp CustomObject=> ActObj
        IF ActObj['value']['ok'] <> True THEN
            SET RowError TO $'''ステップ1（fill #search-key）で要素が見つかりません'''
        END
        IF RowError <> $'''''' THEN
            SET NgCount TO NgCount + 1
            Web.InvokeWebService.InvokeWebService Url: ShotUrl Method: Web.Method.Get Accept: AppJson ContentType: AppJson EncodeRequestBody: False FailOnErrorStatus: False Response=> ShotResp StatusCode=> ShotStatus
            Variables.ConvertJsonToCustomObject Json: ShotResp CustomObject=> ShotObj
            SET ShotB64 TO ShotObj['value']
            File.ConvertFromBase64 Base64Text: ShotB64 File: FailShot IfFileExists: File.IfExists.DoNothing
            File.WriteText File: ResultFile TextToWrite: $'''%RowId%,%RowKey%,失敗,"%RowError%","%FailShot%",%RecStamp%''' AppendNewLine: True IfFileExists: File.IfFileExists.Append
            File.WriteText File: LogFile TextToWrite: $'''[%RecStamp%] %RowId% / %RowKey% 失敗 %RowError%''' AppendNewLine: True IfFileExists: File.IfFileExists.Append
            NEXT LOOP
        END
        WAIT 1

        # [2] click #btn-search
        SET ActBody TO $'''{"script": "%JsAct%", "args": [["#btn-search", "aria/Search"], "click", ""]}'''
        Web.InvokeWebService.InvokeWebService Url: ExecUrl Method: Web.Method.Post Accept: AppJson ContentType: AppJson RequestBody: ActBody EncodeRequestBody: False FailOnErrorStatus: False Response=> ActResp StatusCode=> ActStatus
        Variables.ConvertJsonToCustomObject Json: ActResp CustomObject=> ActObj
        IF ActObj['value']['ok'] <> True THEN
            SET RowError TO $'''ステップ2（click #btn-search）で要素が見つかりません'''
        END
        IF RowError <> $'''''' THEN
            SET NgCount TO NgCount + 1
            Web.InvokeWebService.InvokeWebService Url: ShotUrl Method: Web.Method.Get Accept: AppJson ContentType: AppJson EncodeRequestBody: False FailOnErrorStatus: False Response=> ShotResp StatusCode=> ShotStatus
            Variables.ConvertJsonToCustomObject Json: ShotResp CustomObject=> ShotObj
            SET ShotB64 TO ShotObj['value']
            File.ConvertFromBase64 Base64Text: ShotB64 File: FailShot IfFileExists: File.IfExists.DoNothing
            File.WriteText File: ResultFile TextToWrite: $'''%RowId%,%RowKey%,失敗,"%RowError%","%FailShot%",%RecStamp%''' AppendNewLine: True IfFileExists: File.IfFileExists.Append
            File.WriteText File: LogFile TextToWrite: $'''[%RecStamp%] %RowId% / %RowKey% 失敗 %RowError%''' AppendNewLine: True IfFileExists: File.IfFileExists.Append
            NEXT LOOP
        END
        WAIT 1

        # [3] click aria/%Col2%
        SET ActBody TO $'''{"script": "%JsAct%", "args": [["aria/%Col2%", "#result-link"], "click", ""]}'''
        Web.InvokeWebService.InvokeWebService Url: ExecUrl Method: Web.Method.Post Accept: AppJson ContentType: AppJson RequestBody: ActBody EncodeRequestBody: False FailOnErrorStatus: False Response=> ActResp StatusCode=> ActStatus
        Variables.ConvertJsonToCustomObject Json: ActResp CustomObject=> ActObj
        IF ActObj['value']['ok'] <> True THEN
            SET RowError TO $'''ステップ3（click aria/%Col2%）で要素が見つかりません'''
        END
        IF RowError <> $'''''' THEN
            SET NgCount TO NgCount + 1
            Web.InvokeWebService.InvokeWebService Url: ShotUrl Method: Web.Method.Get Accept: AppJson ContentType: AppJson EncodeRequestBody: False FailOnErrorStatus: False Response=> ShotResp StatusCode=> ShotStatus
            Variables.ConvertJsonToCustomObject Json: ShotResp CustomObject=> ShotObj
            SET ShotB64 TO ShotObj['value']
            File.ConvertFromBase64 Base64Text: ShotB64 File: FailShot IfFileExists: File.IfExists.DoNothing
            File.WriteText File: ResultFile TextToWrite: $'''%RowId%,%RowKey%,失敗,"%RowError%","%FailShot%",%RecStamp%''' AppendNewLine: True IfFileExists: File.IfFileExists.Append
            File.WriteText File: LogFile TextToWrite: $'''[%RecStamp%] %RowId% / %RowKey% 失敗 %RowError%''' AppendNewLine: True IfFileExists: File.IfFileExists.Append
            NEXT LOOP
        END
        WAIT 1

        # [4] click #btn-confirm
        SET ActBody TO $'''{"script": "%JsAct%", "args": [["#btn-confirm", "aria/Confirm", "text/Confirm"], "click", ""]}'''
        Web.InvokeWebService.InvokeWebService Url: ExecUrl Method: Web.Method.Post Accept: AppJson ContentType: AppJson RequestBody: ActBody EncodeRequestBody: False FailOnErrorStatus: False Response=> ActResp StatusCode=> ActStatus
        Variables.ConvertJsonToCustomObject Json: ActResp CustomObject=> ActObj
        IF ActObj['value']['ok'] <> True THEN
            SET RowError TO $'''ステップ4（click #btn-confirm）で要素が見つかりません'''
        END

            Web.InvokeWebService.InvokeWebService Url: ShotUrl Method: Web.Method.Get Accept: AppJson ContentType: AppJson EncodeRequestBody: False FailOnErrorStatus: False Response=> ShotResp StatusCode=> ShotStatus
            Variables.ConvertJsonToCustomObject Json: ShotResp CustomObject=> ShotObj
            SET ShotB64 TO ShotObj['value']
            File.ConvertFromBase64 Base64Text: ShotB64 File: FailShot IfFileExists: File.IfExists.DoNothing
            File.WriteText File: ResultFile TextToWrite: $'''%RowId%,%RowKey%,失敗,"%RowError%","%FailShot%",%RecStamp%''' AppendNewLine: True IfFileExists: File.IfFileExists.Append
            File.WriteText File: LogFile TextToWrite: $'''[%RecStamp%] %RowId% / %RowKey% 失敗 %RowError%''' AppendNewLine: True IfFileExists: File.IfFileExists.Append
            NEXT LOOP
        END
        WAIT 1

        # 💬 確認済み画面をエビデンス保存
        # [5] click #home-link
        SET ActBody TO $'''{"script": "%JsAct%", "args": [["#home-link", "aria/Home"], "click", ""]}'''
        Web.InvokeWebService.InvokeWebService Url: ExecUrl Method: Web.Method.Post Accept: AppJson ContentType: AppJson RequestBody: ActBody EncodeRequestBody: False FailOnErrorStatus: False Response=> ActResp StatusCode=> ActStatus
        Variables.ConvertJsonToCustomObject Json: ActResp CustomObject=> ActObj
        IF ActObj['value']['ok'] <> True THEN
            SET RowError TO $'''ステップ5（click #home-link）で要素が見つかりません'''
        END
        IF RowError <> $'''''' THEN
            SET NgCount TO NgCount + 1
            Web.InvokeWebService.InvokeWebService Url: ShotUrl Method: Web.Method.Get Accept: AppJson ContentType: AppJson EncodeRequestBody: False FailOnErrorStatus: False Response=> ShotResp StatusCode=> ShotStatus

END
