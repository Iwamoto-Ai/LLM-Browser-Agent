# ============================================================
# pad_sample.robin.txt
# Power Automate Desktop から WebDriver を直接叩くバッチ処理のサンプル
#
# 対象      : 同ディレクトリの sample.html（単一ファイルのデモページ）
# 解説      : ../../docs/PAD_WebDriver.md
# ライセンス: Apache-2.0
#
# 貼り付け方: PAD のフローデザイナーでキャンバスをクリックし Ctrl+V
#   ※ Power Fx を有効にしていないフローに貼ってください
#
# 準備:
#   1. sample.html と sample_batch.csv を C:\temp\ にコピー
#   2. Edge と同じメジャーバージョンの msedgedriver.exe を C:\temp\ に置く
#   3. 127.0.0.1 / localhost をプロキシ除外に入れる
#
# このサンプルが示すもの:
#   - WebDriver セッションの開始と終了
#   - 共通 JavaScript による click / fill / exists
#   - 手動ログイン（PAD はパスワードを扱わない）
#   - 明細 CSV のループ、skip 列、件数上限、失敗分の再実行
#   - 失敗時のエビデンス取得と、次の件の前に起点へ戻す復帰処理
#   - ブラウザーのページだけを撮るエビデンス（デスクトップは写らない）
#
# 付属の sample_batch.csv は、1回の実行で成功・失敗・復帰・スキップの
# 4パターンすべてを通るように並べてあります。
#   DEMO-001 / K-1001 → 成功
#   DEMO-002 / K-9999 → 失敗（存在しないキーなので検索0件）
#   DEMO-003 / K-1002 → 復帰して成功  ★ここが成功すれば復帰処理が効いている
#   DEMO-004 / K-1003 → スキップ
# ============================================================

# ---------- 設定：配布時に触るのはここだけ ----------
SET BaseDir TO $'''C:\\temp'''
# ※ 末尾に \ を付けないこと（%BaseDir%\file.png が二重になる）

SET DriverExe TO $'''%BaseDir%\\msedgedriver.exe'''
SET DriverUrl TO $'''http://127.0.0.1:9515'''
SET DetailsFile TO $'''%BaseDir%\\sample_batch.csv'''
SET ResultFile TO $'''%BaseDir%\\pad_sample_result.csv'''
SET LogFile TO $'''%BaseDir%\\pad_sample.log'''
SET ShotDir TO $'''%BaseDir%'''

# file:// はスラッシュ表記なので BaseDir から組めない
SET TargetUrl TO $'''file:///C:/temp/sample.html'''

# ---------- 運用スイッチ ----------
# まず 1 にして 1 件だけ流し、画面と結果を目で確認してから増やす
SET MaxItems TO 10

# 失敗分だけを再実行するとき True。読み込み元と出力先が自動で切り替わる
SET RetryMode TO False

# 完了確認に使う画面上の文字列。空にすると確認をスキップする
SET DoneText TO $'''has been confirmed'''

IF RetryMode THEN
    SET DetailsFile TO $'''%BaseDir%\\pad_sample_result.csv'''
    SET ResultFile TO $'''%BaseDir%\\pad_sample_result_retry.csv'''
END

# ---------- 要素操作の共通 JavaScript ----------

SET JsAct TO $'''var cands = arguments[0], action = arguments[1], value = arguments[2]; var Q = String.fromCharCode(3'''

# ---------- 古いドライバーが残っていたら終了する ----------
# 前回の実行が異常終了すると msedgedriver がポート 9515 を掴んだまま残り、
# 新しいドライバーが起動できない（古い方が応答してしまう）。
System.TerminateProcess.TerminateProcessByName ProcessName: $'''msedgedriver'''
WAIT 1

# ---------- ドライバーを起動 ----------
System.RunApplication.RunApplication ApplicationPath: DriverExe CommandLineArguments: $'''--port=9515''' WindowStyle: System.ProcessWindowStyle.Hidden ProcessId=> DriverPid
WAIT 3

# ---------- セッション開始（ブラウザーが立ち上がる）----------
# EncodeRequestBody: False が必須。既定だと本文が URL エンコードされ、
# WebDriver が invalid argument: missing command parameters を返す。
SET SessionBody TO $'''{"capabilities": {"alwaysMatch": {"browserName": "MicrosoftEdge"}}}'''
SET AppJson TO $'''application/json'''
SET NewUrl TO $'''%DriverUrl%/session'''
Web.InvokeWebService.InvokeWebService Url: NewUrl Method: Web.Method.Post Accept: AppJson ContentType: AppJson RequestBody: SessionBody EncodeRequestBody: False FailOnErrorStatus: False Response=> SessionResp StatusCode=> SessionStatus
# ↓ sessionId が取れないときはこの行を有効にして生の応答を確認する
# Display.ShowMessageDialog.ShowMessage Title: $'''WebDriver 応答''' Message: SessionResp Icon: Display.Icon.Information Buttons: Display.Buttons.OK DefaultButton: Display.DefaultButton.Button1 IsTopMost: True ButtonPressed=> BtnDbg
Variables.ConvertJsonToCustomObject Json: SessionResp CustomObject=> SessionObj
# カスタムオブジェクトはブラケット記法で参照する。ドット記法は使えない。
SET SessionId TO SessionObj['value']['sessionId']
SET ExecUrl TO $'''%DriverUrl%/session/%SessionId%/execute/sync'''
SET GoUrl TO $'''%DriverUrl%/session/%SessionId%/url'''
SET RectUrl TO $'''%DriverUrl%/session/%SessionId%/window/rect'''
SET ShotUrl TO $'''%DriverUrl%/session/%SessionId%/screenshot'''
SET QuitUrl TO $'''%DriverUrl%/session/%SessionId%'''

# ================= セットアップ（最初に 1 回）=================
SET RowError TO $''''''
SET Halt TO False

# ウィンドウサイズ。エビデンスに写る範囲がこれで決まる。
# 複数の PC に配布する場合は、一番小さい画面に収まるサイズにそろえること。
SET RectBody TO $'''{"width": 1600, "height": 1000}'''
Web.InvokeWebService.InvokeWebService Url: RectUrl Method: Web.Method.Post Accept: AppJson ContentType: AppJson RequestBody: RectBody EncodeRequestBody: False FailOnErrorStatus: False Response=> RectResp StatusCode=> RectStatus

# ページを開く
SET UrlBody TO $'''{"url": "%TargetUrl%"}'''
Web.InvokeWebService.InvokeWebService Url: GoUrl Method: Web.Method.Post Accept: AppJson ContentType: AppJson RequestBody: UrlBody EncodeRequestBody: False FailOnErrorStatus: False Response=> UrlResp StatusCode=> UrlStatus
WAIT 2

# ---------- 手動ログイン ----------
# フローを止めて人がログインする。PAD はパスワードを一度も受け取らないので、
# 変数ペイン・実行ログ・エラーメッセージのどこにも残らない。
#
# WebDriver は自分が起動したブラウザーしか操作できない。別に開いてある
# 普段のブラウザーでログインしても、フローはそのタブを見られない。
Display.ShowMessageDialog.ShowMessage Title: $'''手動ログイン''' Message: $'''いま開いたブラウザーでログインし、ホーム画面が出てから[OK]を押してください。ブラウザーは閉じないでください。''' Icon: Display.Icon.Information Buttons: Display.Buttons.OKCancel DefaultButton: Display.DefaultButton.Button1 IsTopMost: True ButtonPressed=> LoginBtn
IF LoginBtn = $'''Cancel''' THEN
    SET Halt TO True
END

# ---------- ループの起点へ移動 ----------
# ループの始点と終点を同じ画面にそろえるための一手。
SET ActBody TO $'''{"script": "%JsAct%", "args": [["#menu-orders", "aria/Orders"], "click", ""]}'''
Web.InvokeWebService.InvokeWebService Url: ExecUrl Method: Web.Method.Post Accept: AppJson ContentType: AppJson RequestBody: ActBody EncodeRequestBody: False FailOnErrorStatus: False Response=> ActResp StatusCode=> ActStatus
Variables.ConvertJsonToCustomObject Json: ActResp CustomObject=> ActObj
IF ActObj['value']['ok'] <> True THEN
    SET RowError TO $'''セットアップ（click Orders）で要素が見つかりません'''
END
WAIT 1

# セットアップに失敗したら明細を 1 件も流さない。
# ここで止めないと、ループ先頭で RowError がリセットされるため
# 全件が「ステップ1で要素が見つかりません」という偽の失敗になる。
IF RowError <> $'''''' THEN
    SET Halt TO True
END

# ================= 明細を読み込む =================
IF Halt = False THEN
    File.ReadFromCSVFile.ReadCSV CSVFile: DetailsFile Encoding: File.CSVEncoding.UTF8 TrimFields: True FirstLineContainsColumnNames: True ColumnsSeparator: File.CSVColumnsSeparator.Comma CSVTable=> Rows
    # 結果 CSV は、そのまま明細として読み直せる列構成にする。
    # ID だけでなく業務キーも入れておかないと再実行で対象を特定できない。
    File.WriteText File: ResultFile TextToWrite: $'''ID,KEY,結果,理由,エビデンス,実行日時''' AppendNewLine: True IfFileExists: File.IfFileExists.Overwrite
    File.WriteText File: LogFile TextToWrite: $'''===== 実行開始 上限%MaxItems%件 RetryMode=%RetryMode% =====''' AppendNewLine: True IfFileExists: File.IfFileExists.Append
    SET OkCount TO 0
    SET NgCount TO 0
    SET SkipCount TO 0
    SET Attempted TO 0
    SET PrevFailed TO False

    # ================= 明細ごとの繰り返し =================
    LOOP FOREACH Row IN Rows
        # 列は変数に取り出す（リテラル内に単引用符を書かないため）
        SET RowId TO Row['ID']
        SET RowKey TO Row['KEY']
        SET RowError TO $''''''

        # 再実行モードでは「失敗」行だけを対象にする
        IF RetryMode THEN
            IF Row['結果'] <> $'''失敗''' THEN
                NEXT LOOP
            END
        END

        # skip 列に値がある行は飛ばす（再実行 CSV には skip 列が無いので除外）
        IF RetryMode = False THEN
            IF Row['skip'] <> $'''''' THEN
                SET SkipCount TO SkipCount + 1
                File.WriteText File: ResultFile TextToWrite: $'''%RowId%,%RowKey%,スキップ,,,''' AppendNewLine: True IfFileExists: File.IfFileExists.Append
                NEXT LOOP
            END
        END

        # 件数上限。打ち切った行も記録に残す（どこから再開するか分かる）
        IF Attempted >= MaxItems THEN
            File.WriteText File: ResultFile TextToWrite: $'''%RowId%,%RowKey%,未実行,"件数上限 %MaxItems% 件に達したため",,''' AppendNewLine: True IfFileExists: File.IfFileExists.Append
            NEXT LOOP
        END
        SET Attempted TO Attempted + 1

        # 直前の件が失敗していたら、起点に復帰してから始める。
        # 失敗した画面のまま次の件を始めると 1 件の失敗が全件に連鎖する。
        IF PrevFailed THEN
            SET ActBody TO $'''{"script": "%JsAct%", "args": [["#home-link", "aria/Home"], "click", ""]}'''
            Web.InvokeWebService.InvokeWebService Url: ExecUrl Method: Web.Method.Post Accept: AppJson ContentType: AppJson RequestBody: ActBody EncodeRequestBody: False FailOnErrorStatus: False Response=> ActResp StatusCode=> ActStatus
            WAIT 1
            SET ActBody TO $'''{"script": "%JsAct%", "args": [["#menu-orders", "aria/Orders"], "click", ""]}'''
            Web.InvokeWebService.InvokeWebService Url: ExecUrl Method: Web.Method.Post Accept: AppJson ContentType: AppJson RequestBody: ActBody EncodeRequestBody: False FailOnErrorStatus: False Response=> ActResp StatusCode=> ActStatus
            WAIT 1
            SET PrevFailed TO False
        END
        # 悲観的に置く。最後まで通れば False に戻る。
        # こうするとフラグを立てる箇所が 1 行で済む。
        SET PrevFailed TO True

        # 日時つきのファイル名を作る。
        # MM は月、mm は分。yyyymmdd と書くと月の位置に分が入る。
        DateTime.GetCurrentDateTime.Local DateTimeFormat: DateTime.DateTimeFormat.DateAndTime CurrentDateTime=> NowDt
        Text.ConvertDateTimeToText.FromCustomDateTime DateTime: NowDt CustomFormat: $'''yyyyMMdd_HHmmss''' Result=> Stamp
        Text.ConvertDateTimeToText.FromCustomDateTime DateTime: NowDt CustomFormat: $'''yyyy/MM/dd HH:mm:ss''' Result=> RecStamp
        SET ShotPath TO $'''%ShotDir%\\%RowId%__%RowKey%__%Stamp%.png'''
        SET FailShot TO $'''%ShotDir%\\fail__%RowId%__%RowKey%__%Stamp%.png'''

        File.WriteText File: LogFile TextToWrite: $'''[%RecStamp%] %RowId% / %RowKey% 開始''' AppendNewLine: True IfFileExists: File.IfFileExists.Append

        # ---------- [1] 検索キーを入力 ----------
        SET ActBody TO $'''{"script": "%JsAct%", "args": [["#search-key", "aria/Order key"], "fill", "%RowKey%"]}'''
        Web.InvokeWebService.InvokeWebService Url: ExecUrl Method: Web.Method.Post Accept: AppJson ContentType: AppJson RequestBody: ActBody EncodeRequestBody: False FailOnErrorStatus: False Response=> ActResp StatusCode=> ActStatus
        Variables.ConvertJsonToCustomObject Json: ActResp CustomObject=> ActObj
        IF ActObj['value']['ok'] <> True THEN
            SET RowError TO $'''ステップ1（fill #search-key）で要素が見つかりません'''
        END
        IF RowError <> $'''''' THEN
            SET NgCount TO NgCount + 1
            Web.InvokeWebService.InvokeWebService Url: ShotUrl Method: Web.Method.Get Accept: AppJson ContentType: AppJson EncodeRequestBody: False FailOnErrorStatus: False Response=> ShotResp StatusCode=> ShotStatus
            Variables.ConvertJsonToCustomObject Json: ShotResp CustomObject=> ShotObj
            SET ShotB64 TO ShotObj['value']
            File.ConvertFromBase64 Base64Text: ShotB64 File: FailShot IfFileExists: File.IfExists.DoNothing
            File.WriteText File: ResultFile TextToWrite: $'''%RowId%,%RowKey%,失敗,"%RowError%","%FailShot%",%RecStamp%''' AppendNewLine: True IfFileExists: File.IfFileExists.Append
            File.WriteText File: LogFile TextToWrite: $'''[%RecStamp%] %RowId% / %RowKey% 失敗 %RowError%''' AppendNewLine: True IfFileExists: File.IfFileExists.Append
            NEXT LOOP
        END
        WAIT 1

        # ---------- [2] 検索実行 ----------
        SET ActBody TO $'''{"script": "%JsAct%", "args": [["#btn-search", "aria/Search"], "click", ""]}'''
        Web.InvokeWebService.InvokeWebService Url: ExecUrl Method: Web.Method.Post Accept: AppJson ContentType: AppJson RequestBody: ActBody EncodeRequestBody: False FailOnErrorStatus: False Response=> ActResp StatusCode=> ActStatus
        Variables.ConvertJsonToCustomObject Json: ActResp CustomObject=> ActObj
        IF ActObj['value']['ok'] <> True THEN
            SET RowError TO $'''ステップ2（click #btn-search）で要素が見つかりません'''
        END
        IF RowError <> $'''''' THEN
            SET NgCount TO NgCount + 1
            Web.InvokeWebService.InvokeWebService Url: ShotUrl Method: Web.Method.Get Accept: AppJson ContentType: AppJson EncodeRequestBody: False FailOnErrorStatus: False Response=> ShotResp StatusCode=> ShotStatus
            Variables.ConvertJsonToCustomObject Json: ShotResp CustomObject=> ShotObj
            SET ShotB64 TO ShotObj['value']
            File.ConvertFromBase64 Base64Text: ShotB64 File: FailShot IfFileExists: File.IfExists.DoNothing
            File.WriteText File: ResultFile TextToWrite: $'''%RowId%,%RowKey%,失敗,"%RowError%","%FailShot%",%RecStamp%''' AppendNewLine: True IfFileExists: File.IfFileExists.Append
            File.WriteText File: LogFile TextToWrite: $'''[%RecStamp%] %RowId% / %RowKey% 失敗 %RowError%''' AppendNewLine: True IfFileExists: File.IfFileExists.Append
            NEXT LOOP
        END
        WAIT 1

        # ---------- [3] 検索結果のリンクを開く ----------
        # セレクタに明細の値を埋める。存在しないキーだと行が生成されないため、
        # ここで ok: false になる（= 検索 0 件の検出）。
        SET ActBody TO $'''{"script": "%JsAct%", "args": [["aria/%RowKey%", "#result-link"], "click", ""]}'''
        Web.InvokeWebService.InvokeWebService Url: ExecUrl Method: Web.Method.Post Accept: AppJson ContentType: AppJson RequestBody: ActBody EncodeRequestBody: False FailOnErrorStatus: False Response=> ActResp StatusCode=> ActStatus
        Variables.ConvertJsonToCustomObject Json: ActResp CustomObject=> ActObj
        IF ActObj['value']['ok'] <> True THEN
            SET RowError TO $'''ステップ3（click aria/%RowKey%）該当データが見つかりません'''
        END
        IF RowError <> $'''''' THEN
            SET NgCount TO NgCount + 1
            Web.InvokeWebService.InvokeWebService Url: ShotUrl Method: Web.Method.Get Accept: AppJson ContentType: AppJson EncodeRequestBody: False FailOnErrorStatus: False Response=> ShotResp StatusCode=> ShotStatus
            Variables.ConvertJsonToCustomObject Json: ShotResp CustomObject=> ShotObj
            SET ShotB64 TO ShotObj['value']
            File.ConvertFromBase64 Base64Text: ShotB64 File: FailShot IfFileExists: File.IfExists.DoNothing
            File.WriteText File: ResultFile TextToWrite: $'''%RowId%,%RowKey%,失敗,"%RowError%","%FailShot%",%RecStamp%''' AppendNewLine: True IfFileExists: File.IfFileExists.Append
            File.WriteText File: LogFile TextToWrite: $'''[%RecStamp%] %RowId% / %RowKey% 失敗 %RowError%''' AppendNewLine: True IfFileExists: File.IfFileExists.Append
            NEXT LOOP
        END
        WAIT 1

        # ---------- [4] 確認を実行 ----------
        SET ActBody TO $'''{"script": "%JsAct%", "args": [["#btn-confirm", "aria/Confirm", "text/Confirm"], "click", ""]}'''
        Web.InvokeWebService.InvokeWebService Url: ExecUrl Method: Web.Method.Post Accept: AppJson ContentType: AppJson RequestBody: ActBody EncodeRequestBody: False FailOnErrorStatus: False Response=> ActResp StatusCode=> ActStatus
        Variables.ConvertJsonToCustomObject Json: ActResp CustomObject=> ActObj
        IF ActObj['value']['ok'] <> True THEN
            SET RowError TO $'''ステップ4（click #btn-confirm）で要素が見つかりません'''
        END
        IF RowError <> $'''''' THEN
            SET NgCount TO NgCount + 1
            Web.InvokeWebService.InvokeWebService Url: ShotUrl Method: Web.Method.Get Accept: AppJson ContentType: AppJson EncodeRequestBody: False FailOnErrorStatus: False Response=> ShotResp StatusCode=> ShotStatus
            Variables.ConvertJsonToCustomObject Json: ShotResp CustomObject=> ShotObj
            SET ShotB64 TO ShotObj['value']
            File.ConvertFromBase64 Base64Text: ShotB64 File: FailShot IfFileExists: File.IfExists.DoNothing
            File.WriteText File: ResultFile TextToWrite: $'''%RowId%,%RowKey%,失敗,"%RowError%","%FailShot%",%RecStamp%''' AppendNewLine: True IfFileExists: File.IfFileExists.Append
            File.WriteText File: LogFile TextToWrite: $'''[%RecStamp%] %RowId% / %RowKey% 失敗 %RowError%''' AppendNewLine: True IfFileExists: File.IfFileExists.Append
            NEXT LOOP
        END
        WAIT 1

        # ---------- [5] 完了確認 ----------
        # 「ボタンは押せたが実際には処理されていない」を検知する。
        # DoneText が空なら何もしない。
        IF DoneText <> $'''''' THEN
            SET ActBody TO $'''{"script": "%JsAct%", "args": [[], "exists", "%DoneText%"]}'''
            Web.InvokeWebService.InvokeWebService Url: ExecUrl Method: Web.Method.Post Accept: AppJson ContentType: AppJson RequestBody: ActBody EncodeRequestBody: False FailOnErrorStatus: False Response=> ActResp StatusCode=> ActStatus
            Variables.ConvertJsonToCustomObject Json: ActResp CustomObject=> ActObj
            IF ActObj['value']['ok'] <> True THEN
                SET RowError TO $'''ステップ5（完了確認）画面に「%DoneText%」が見つかりません'''
            END
        END
        IF RowError <> $'''''' THEN
            SET NgCount TO NgCount + 1
            Web.InvokeWebService.InvokeWebService Url: ShotUrl Method: Web.Method.Get Accept: AppJson ContentType: AppJson EncodeRequestBody: False FailOnErrorStatus: False Response=> ShotResp StatusCode=> ShotStatus
            Variables.ConvertJsonToCustomObject Json: ShotResp CustomObject=> ShotObj
            SET ShotB64 TO ShotObj['value']
            File.ConvertFromBase64 Base64Text: ShotB64 File: FailShot IfFileExists: File.IfExists.DoNothing
            File.WriteText File: ResultFile TextToWrite: $'''%RowId%,%RowKey%,失敗,"%RowError%","%FailShot%",%RecStamp%''' AppendNewLine: True IfFileExists: File.IfFileExists.Append
            File.WriteText File: LogFile TextToWrite: $'''[%RecStamp%] %RowId% / %RowKey% 失敗 %RowError%''' AppendNewLine: True IfFileExists: File.IfFileExists.Append
            NEXT LOOP
        END

        # ---------- エビデンス保存 ----------
        # WebDriver に撮らせるとページの表示領域だけが写る。
        # フォーカスに依存しないので、実行中に通知が前面に出ても影響しない。
        # 逆にスクロールしないと見えない範囲は写らない点に注意。
        Web.InvokeWebService.InvokeWebService Url: ShotUrl Method: Web.Method.Get Accept: AppJson ContentType: AppJson EncodeRequestBody: False FailOnErrorStatus: False Response=> ShotResp StatusCode=> ShotStatus
        Variables.ConvertJsonToCustomObject Json: ShotResp CustomObject=> ShotObj
        SET ShotB64 TO ShotObj['value']
        # 列挙型の名前空間がアクションごとに違う。File.WriteText は
        # File.IfFileExists.* だが、こちらは File.IfExists.* になる。
        File.ConvertFromBase64 Base64Text: ShotB64 File: ShotPath IfFileExists: File.IfExists.DoNothing

        # ---------- [6] 起点へ戻る ----------
        SET ActBody TO $'''{"script": "%JsAct%", "args": [["#home-link", "aria/Home"], "click", ""]}'''
        Web.InvokeWebService.InvokeWebService Url: ExecUrl Method: Web.Method.Post Accept: AppJson ContentType: AppJson RequestBody: ActBody EncodeRequestBody: False FailOnErrorStatus: False Response=> ActResp StatusCode=> ActStatus
        Variables.ConvertJsonToCustomObject Json: ActResp CustomObject=> ActObj
        IF ActObj['value']['ok'] <> True THEN
            SET RowError TO $'''ステップ6（click Home）で要素が見つかりません'''
        END
        IF RowError <> $'''''' THEN
            SET NgCount TO NgCount + 1
            File.WriteText File: ResultFile TextToWrite: $'''%RowId%,%RowKey%,失敗,"%RowError%","%ShotPath%",%RecStamp%''' AppendNewLine: True IfFileExists: File.IfFileExists.Append
            NEXT LOOP
        END
        WAIT 1

        # ---------- [7] 次の件の起点へ ----------
        SET ActBody TO $'''{"script": "%JsAct%", "args": [["#menu-orders", "aria/Orders"], "click", ""]}'''
        Web.InvokeWebService.InvokeWebService Url: ExecUrl Method: Web.Method.Post Accept: AppJson ContentType: AppJson RequestBody: ActBody EncodeRequestBody: False FailOnErrorStatus: False Response=> ActResp StatusCode=> ActStatus
        Variables.ConvertJsonToCustomObject Json: ActResp CustomObject=> ActObj
        IF ActObj['value']['ok'] <> True THEN
            SET RowError TO $'''ステップ7（click Orders）で要素が見つかりません'''
        END
        IF RowError <> $'''''' THEN
            SET NgCount TO NgCount + 1
            File.WriteText File: ResultFile TextToWrite: $'''%RowId%,%RowKey%,失敗,"%RowError%","%ShotPath%",%RecStamp%''' AppendNewLine: True IfFileExists: File.IfFileExists.Append
            NEXT LOOP
        END
        WAIT 1

        # ---------- ここまで来たら成功 ----------
        SET PrevFailed TO False
        SET OkCount TO OkCount + 1
        File.WriteText File: ResultFile TextToWrite: $'''%RowId%,%RowKey%,成功,,"%ShotPath%",%RecStamp%''' AppendNewLine: True IfFileExists: File.IfFileExists.Append
        File.WriteText File: LogFile TextToWrite: $'''[%RecStamp%] %RowId% / %RowKey% 成功''' AppendNewLine: True IfFileExists: File.IfFileExists.Append
    END
END
